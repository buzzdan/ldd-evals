```go
if parsed.Version != "" && d.Version != parsed.Version {
    d.Version = parsed.Version
    changed = true
}
if len(parsed.Tags) > 0 {
    d.Tags = parsed.Tags
    changed = true
}
d.LastSeen = time.Now()

if changed || force {
    if err := s.repo.Save(context.Background(), d); err != nil {
        return HeartbeatResult{ID: parsed.ID}, fmt.Errorf("save %s: %w", cacheKey(tenant, parsed.ID), err)
    }
    s.mu.Lock()
    s.cache[cacheKey(tenant, parsed.ID)] = d
    s.mu.Unlock()
}
s.lastSeen[parsed.ID] = d.LastSeen
```

A reader cannot predict the story top-down; they must parse conditional persistence and cache locking inline.

- ❌ **Concepts have boxes:** Persistence and version/tag updating lack their own named helpers
